cmake_minimum_required(VERSION 3.16)
project(MiniDiDi VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8 /W4)
endif()
find_package(Threads REQUIRED)
add_executable(minididi server/main.cpp)
target_include_directories(minididi PRIVATE server third_party)
target_link_libraries(minididi PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(minididi PRIVATE ws2_32 crypt32)
endif()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE server third_party)
enable_testing()
add_test(NAME core COMMAND core_tests)
